% Shared preamble of every article under articles/. A package, a macro or a marker that every
% article needs lives here, so it means the same thing everywhere. Notation that belongs to ONE
% article (its symbols, its macros) goes in that article's own main.tex, after this \input.
%
% Each article's main.tex starts with its \documentclass, then \input{../common/preamble}.

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[a4paper, top=20mm, bottom=22mm, left=17mm, right=17mm, columnsep=7mm]{geometry}
\usepackage{amsmath, amssymb}
\usepackage{booktabs, tabularx, array}
\usepackage{enumitem}
\usepackage{xcolor}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, calc}
\usepackage[font=small, labelfont=bf]{caption}
\usepackage{cite}
\usepackage[hidelinks]{hyperref}
\usepackage{xurl}

\setlist{itemsep=1pt, topsep=3pt, leftmargin=*}
\newcolumntype{L}{>{\raggedright\arraybackslash}X}
\renewcommand{\arraystretch}{1.2}

% ---- markers ------------------------------------------------------------------------------
% \todo{...} marks text that is still to be written; grep for it before sharing the draft.
\newcommand{\todo}[1]{{\color{red!60!black}\textbf{[TODO: #1]}}}

% \openissue{...} sets off a question the sources do not settle.
\newcommand{\openissue}[1]{%
  \par\smallskip\noindent
  \fcolorbox{red!60!black}{red!5}{\parbox{\dimexpr\columnwidth-2\fboxsep-2\fboxrule\relax}{%
    \small\textbf{Open.} #1}}\par\smallskip}

% \intake{I-12} tags a statement with the intake line it comes from (the definition article);
% \inferred marks a statement Claude inferred, which the user must confirm or strike.
\newcommand{\intake}[1]{\textsuperscript{\color{gray}[#1]}}
\newcommand{\inferred}{\textsuperscript{\color{orange!70!black}[inferred]}}

% \statusbox: version and review status, printed under the abstract. \articleversion and
% \articlereviewstatus come from the generated sections/versions.tex, which every article
% inputs after this preamble.
\newcommand{\statusbox}{\par\noindent
  \fcolorbox{red!60!black}{red!5}{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{\small
    \textbf{Version \articleversion, \articlereviewstatus.} Written with an AI assistant
    (Claude); check each claim against its source before relying on it.}}\par\bigskip}

% \tbdnote: says what TBD means. Printed under the status box of every article that writes TBD.
\newcommand{\tbdnote}{\par\noindent{\small\textbf{TBD} (to be determined) marks a value or
  decision nobody can give yet. It names an owner and, where one is set, a resolve-by date.}\par\bigskip}
